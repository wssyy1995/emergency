    <script>
        // Function to trigger the animation
        function triggerAnimation() {
            const container = document.getElementById('animationContainer');
            const checkmark = document.getElementById('checkmark');
            const glow = document.getElementById('glow');
            const particles = [
                document.getElementById('particle1'),
                document.getElementById('particle2'),
                document.getElementById('particle3'),
                document.getElementById('particle4')
            ];

            // Reset animation states (if triggered multiple times)
            checkmark.style.animation = 'none';
            glow.style.animation = 'none';
            particles.forEach(p => p.style.animation = 'none');
            void checkmark.offsetWidth; // Trigger reflow to restart animation
            void glow.offsetWidth;
            particles.forEach(p => void p.offsetWidth);

            // Apply animations
            checkmark.style.animation = 'fadeInScale 0.5s ease-out forwards';
            glow.style.animation = 'glowPulse 1.5s ease-out forwards';
            particles.forEach((p, index) => {
                p.style.animation = `particleFadeOut 1.5s ease-out forwards ${index * 0.1}s`;
            });

             // Optional: Hide elements after animation completes to clean up
            setTimeout(() => {
                checkmark.style.opacity = '0';
                checkmark.style.transform = 'scale(0.5)'; // Reset for next trigger
                glow.style.opacity = '0';
                glow.style.boxShadow = 'none';
                particles.forEach(p => {
                    p.style.opacity = '0';
                    p.style.transform = 'translateY(0) scale(1)'; // Reset for next trigger
                });
            }, 1500);
        }

        // Trigger animation when the page loads (for demonstration)
        document.addEventListener('DOMContentLoaded', () => {
            triggerAnimation();
            // You might want to trigger it on a button click in your actual game
            // For demo, re-trigger every 3 seconds
            setInterval(triggerAnimation, 3000);
        });
    </script>
